      }

      function openDrawerForCard(card){
        if(!card || !drawer) return;

        setActiveCard(card);

        const title = (card.querySelector('.de-sp4-cardtitle')?.textContent || 'Details').trim();
        const phase = (card.getAttribute('data-phase') || 'Details').trim();
        const details = card.querySelector('.de-sp4-details');

        drawerTitle.textContent = title;
        drawerTag.textContent = phase;
        drawerBody.innerHTML = details ? details.innerHTML : '<p>Add a hidden <b>.de-sp4-details</b> block to this card.</p>';
        drawer.hidden = false;

        // update relevant progress
        const scope = card.getAttribute('data-scope') || 'track';
        if(scope === 'lead') updateLeadProgress();
        if(scope === 'track') updateTrackProgress();

        setTimeout(() => drawer.scrollIntoView({ behavior: 'smooth', block: 'nearest' }), 60);
      }

      function closeDrawer(){ if(drawer) drawer.hidden = true; }

      function activateTab(key){
        tabs.forEach(t => {
          const on = t.dataset.tab === key;
          t.classList.toggle('is-active', on);
          t.setAttribute('aria-selected', on ? 'true' : 'false');
        });

        panels.forEach(p => {
          const on = p.dataset.panel === key;
          p.hidden = !on;
          p.classList.toggle('is-active', on);
        });

        const activePanel = getActivePanel();
        if(activePanel){
          const first = activePanel.querySelector('[data-card][data-scope="track"]');
          if(first){
            activePanel.querySelectorAll('[data-card][data-scope="track"].is-active').forEach(c=>c.classList.remove('is-active'));
            first.classList.add('is-active');
          }
        }

        closeDrawer();
        updateTrackProgress();
        runSearchFilter();
      }

      // Accordion toggle
      function toggleAcc(acc, forceOpen){
        if(!acc) return;
        const body = acc.querySelector('[data-acc-body]');
        const isOpen = acc.classList.contains('is-open');
        const nextOpen = (typeof forceOpen === 'boolean') ? forceOpen : !isOpen;
        acc.classList.toggle('is-open', nextOpen);
        if(body) body.hidden = !nextOpen;
      }

      // Show only one (collapse the other)
      function showOnly(which){
        const leadOn = which === 'lead';
        const trackOn = which === 'track';
        toggleAcc(leadAcc, leadOn);
        toggleAcc(trackAcc, trackOn);

        // pill visual state
        pills.forEach(p => p.classList.remove('is-on'));
        if(leadOn) root.querySelector('.de-sp4-pill[data-view="single-lead"]')?.classList.add('is-on');
        if(trackOn) root.querySelector('.de-sp4-pill[data-view="single-track"]')?.classList.add('is-on');

        // also update the "Lead Gen" and "Track" pills to reflect real state
        root.querySelector('.de-sp4-pill[data-view="lead"]')?.classList.toggle('is-on', leadOn);
        root.querySelector('.de-sp4-pill[data-view="track"]')?.classList.toggle('is-on', trackOn);

        closeDrawer();
      }

      // Search should filter Lead Gen + active track panel (even if one is collapsed)
      function runSearchFilter(){
        const q = (searchInput?.value || '').toLowerCase().trim();

        const leadCards = Array.from(root.querySelectorAll('[data-timeline="lead"] [data-card][data-scope="lead"]'));
        const activePanel = getActivePanel();
        const panelCards = activePanel ? Array.from(activePanel.querySelectorAll('[data-card][data-scope="track"]')) : [];

        const all = leadCards.concat(panelCards);

        all.forEach(card => {
          const keywords = (card.getAttribute('data-keywords') || '').toLowerCase();
          const title = (card.querySelector('.de-sp4-cardtitle')?.textContent || '').toLowerCase();
          const text = (card.textContent || '').toLowerCase();

          const match = !q || keywords.includes(q) || title.includes(q) || text.includes(q);
          card.style.opacity = match ? '1' : '0.35';
          card.style.transform = match ? '' : 'scale(0.98)';
          card.style.pointerEvents = match ? '' : 'none';
        });
      }

      // Wire accordion heads
      accs.forEach(acc => {
        const head = acc.querySelector('[data-acc-toggle]');
        const body = acc.querySelector('[data-acc-body]');
        if(body) body.hidden = !acc.classList.contains('is-open');
        if(head){
          head.addEventListener('click', () => {
            toggleAcc(acc);
            // keep "Lead/Track" pills synced with actual open state
            const leadOpen = leadAcc?.classList.contains('is-open');
            const trackOpen = trackAcc?.classList.contains('is-open');
            root.querySelector('.de-sp4-pill[data-view="lead"]')?.classList.toggle('is-on', !!leadOpen);
            root.querySelector('.de-sp4-pill[data-view="track"]')?.classList.toggle('is-on', !!trackOpen);
            closeDrawer();
          });
        }
      });

      // View pills actions
      pills.forEach(p => {
        p.addEventListener('click', () => {
          const mode = p.dataset.view;

          if(mode === 'lead' || mode === 'track'){
            // toggle independently
            const targetAcc = mode === 'lead' ? leadAcc : trackAcc;
            toggleAcc(targetAcc);
            const leadOpen = leadAcc?.classList.contains('is-open');
            const trackOpen = trackAcc?.classList.contains('is-open');
            root.querySelector('.de-sp4-pill[data-view="lead"]')?.classList.toggle('is-on', !!leadOpen);
            root.querySelector('.de-sp4-pill[data-view="track"]')?.classList.toggle('is-on', !!trackOpen);

            // if both ended up closed, reopen track by default
            if(!leadOpen && !trackOpen){
              toggleAcc(trackAcc, true);
              root.querySelector('.de-sp4-pill[data-view="track"]')?.classList.add('is-on');
            }

            closeDrawer();
            return;
          }

          if(mode === 'single-lead'){ showOnly('lead'); return; }
          if(mode === 'single-track'){ showOnly('track'); return; }
        });
      });

      // Tabs
      tabs.forEach(t => t.addEventListener('click', () => activateTab(t.dataset.tab)));

      if(searchInput) searchInput.addEventListener('input', runSearchFilter);

      // Event delegation for details
      root.addEventListener('click', (e) => {
        const closeBtn = e.target.closest('[data-drawer-close]');
        if(closeBtn){ e.preventDefault(); closeDrawer(); return; }

        const toggleBtn = e.target.closest('[data-toggle]');
        if(toggleBtn){
          e.preventDefault();
          const card = toggleBtn.closest('[data-card]');
          openDrawerForCard(card);
          return;
        }

        // click card opens drawer unless selecting text / clicking links
        const card = e.target.closest('[data-card]');
        if(card && !window.getSelection()?.toString()){
          const isLink = e.target.closest('a');
          if(isLink) return;
          openDrawerForCard(card);
        }
      }, true);

      // Esc closes
      document.addEventListener('keydown', (e) => { if(e.key === 'Escape') closeDrawer(); });

      // Init
      activateTab('ecosystem');
      updateLeadProgress();
      updateTrackProgress();

      // Default: show both sections (you can change to showOnly('track') if you prefer)
      toggleAcc(leadAcc, true);
      toggleAcc(trackAcc, true);

      // Sync pills initial state
      root.querySelector('.de-sp4-pill[data-view="lead"]')?.classList.add('is-on');
      root.querySelector('.de-sp4-pill[data-view="track"]')?.classList.add('is-on');

      runSearchFilter();
    })();
  </script>
</section>
