        document.addEventListener('DOMContentLoaded', function() {
            const featureRows = document.querySelectorAll('.feature-row');
            
            featureRows.forEach(row => {
                row.addEventListener('click', function() {
                    const isExpanded = this.classList.contains('expanded');
                    const description = this.getAttribute('data-description');
                    
                    // Close all other expanded rows
                    document.querySelectorAll('.feature-row.expanded').forEach(expandedRow => {
                        if (expandedRow !== this) {
                            expandedRow.classList.remove('expanded');
                            const nextRow = expandedRow.nextElementSibling;
                            if (nextRow && nextRow.classList.contains('description-row')) {
                                nextRow.classList.remove('show');
                                setTimeout(() => nextRow.remove(), 300);
                            }
                        }
                    });
                    
                    if (isExpanded) {
                        // Collapse this row
                        this.classList.remove('expanded');
                        const nextRow = this.nextElementSibling;
                        if (nextRow && nextRow.classList.contains('description-row')) {
                            nextRow.classList.remove('show');
                            setTimeout(() => nextRow.remove(), 300);
                        }
                    } else {
                        // Expand this row
                        this.classList.add('expanded');
                        
                        // Create and insert description row
                        const descRow = document.createElement('tr');
                        descRow.className = 'description-row';
                        descRow.innerHTML = `
                            <td>
                                <div class="description-content">
                                    ${description}
                                </div>
                            </td>
                        `;
                        
                        this.parentNode.insertBefore(descRow, this.nextSibling);
                        setTimeout(() => descRow.classList.add('show'), 10);
                    }
                });
            });
        });
    </script>
