          document.querySelectorAll('#securityInternalTable .feature-row.expanded').forEach(openRow => {

              const r = openRow.nextElementSibling;
              if (r && r.classList.contains('description-row')) r.classList.remove('show');
            }
          });

          if (!descRow || !descRow.classList.contains('description-row')) return;

          if (isExpanded) {
            this.classList.remove('expanded');
            descRow.classList.remove('show');
          } else {
            this.classList.add('expanded');
            const content = descRow.querySelector('.description-content');
            if (content) content.innerHTML = desc || '<p><strong>No internal notes added yet.</strong></p>';
            descRow.classList.add('show');
          }
        });
      });
