  switch(card.autoCalc){

    case 'endpoint': return endpoints;
    case 'site': return sites;
    case 'mobile': return mobile;
    case 'storage': return users*storage;

  }

// === Filtering/Validation ===
function shouldShowCard(card){
  if(searchTerm){
    const q = searchTerm.toLowerCase();
    const hay = [card.title, card.note||'', ...(card.tags||[])].join(' ').toLowerCase();
    if(!hay.includes(q)) return false;
  }
  if(currentFilter==='all') return true;
  const qty=card.qty||0, cost=card.cost||0, price=card.price||0, profit=(price-cost)*qty;
  if(currentFilter==='profit') return profit>0;
  if(currentFilter==='loss') return profit<0;
  if(currentFilter==='incomplete') return (card.cost==null || card.price==null || !card.qty);
  return true;
}

function validateCards(){

    alert.classList.add('active');
  } else { alert.classList.remove('active'); }
}

// === Drag & Drop for lanes (improved, with autoscroll + fallback to Quote) ===
let lastDragScroll = 0;
function wireDnD(){
  Object.entries(laneEls).forEach(([laneId, lane])=>{
    lane.addEventListener('dragover', e=>{
      e.preventDefault();
      let valid=false;
      // Try vendor payload first
      const json = e.dataTransfer.getData('application/json');
      if(json){ try{ const p=JSON.parse(json); if(p.type==='vendor'){ valid = canDropCategoryInLane(p.category, laneId); } } catch(_){} }
      // Fallback to moving a card
      if(!valid && e.dataTransfer.types && Array.from(e.dataTransfer.types).includes('text/plain')){
        const draggedId = e.dataTransfer.getData('text/plain');
        const draggedCard = cards.find(c=>c.id===draggedId);
        valid = draggedCard && canDropInLane(draggedCard, laneId);
      }
      lane.classList.toggle('dragover', valid);
      lane.classList.toggle('reject-drop', !valid);
      e.dataTransfer.dropEffect = valid?'copyMove':'none';

      // Simple autoscroll near viewport edges
      const now = Date.now();
      if(now - lastDragScroll > 50){
        const threshold = 80;
        if(e.clientY < threshold) window.scrollBy(0, -40);
        else if(window.innerHeight - e.clientY < threshold) window.scrollBy(0, 40);
        lastDragScroll = now;
      }
    });
    lane.addEventListener('dragleave', e=>{ if(e.target===lane){ lane.classList.remove('dragover','reject-drop'); }});
    lane.addEventListener('drop', e=>{
      e.preventDefault(); lane.classList.remove('dragover','reject-drop');
      const json = e.dataTransfer.getData('application/json');
      if(json){ try{
        const p = JSON.parse(json);
        if(p.type==='vendor'){
          // If lane rejects, fallback to quote and notify
          const targetLane = canDropCategoryInLane(p.category, laneId) ? laneId : 'quote';
          if(targetLane==='quote' && laneId!=='quote') toast(`That lane doesn't accept “${p.category}”. Added to Quote.`);
          const template = vendorChipToCard(p, targetLane);
          if(template){ cards.push(template); save(); render(); return; }
        }
      }catch(_){ /* ignore */ }
      }
      // else card move
      const id = e.dataTransfer.getData('text/plain');
      const c = cards.find(x=>x.id===id);
      if(c){
        if(canDropInLane(c, laneId)){ c.section = laneId; save(); render(); }
        else { c.section = 'quote'; save(); render(); toast(`Moved to Quote (lane rejects category: ${c.category}).`); }
      }
    });
  });
}

function canDropCategoryInLane(category, laneId){
  if(laneId==='quote') return true;
  const lane = laneEls[laneId]; if(!lane) return false;
  const accepts = (lane.dataset.accepts||'').split(',');
  return accepts.includes(category);
}
function canDropInLane(card, laneId){ return canDropCategoryInLane(card.category, laneId); }

function vendorChipToCard(payload, laneId){
  if(!canDropCategoryInLane(payload.category, laneId)) laneId = 'quote';
  const card = {
    id: generateId(),
    title: `${payload.title}`,
    category: payload.category,
    cost: null,
    price: null,
    qty: 1,
    unit: payload.unit||'user/mo',
    tags: [payload.vendor, ...(payload.tags||[])],
    note: payload.note||'',
    section: laneId,
    autoCalc: guessAutoCalcFromUnit(payload.unit||'user/mo')
  };
  const match = DEFAULT_CATALOG.find(x => x.title.toLowerCase()===card.title.toLowerCase() || (x.category===card.category && x.tags && x.tags.some(t=>card.tags.includes(t))));
  if(match){ card.unit = match.unit; card.autoCalc = match.autoCalc; card.cost = match.cost; card.price = match.price; }
  card.qty = getAutoCalculatedQty(card);
  return card;
}

function guessAutoCalcFromUnit(unit){
  if(unit.includes('user')) return 'user';
  if(unit.includes('endpoint')) return 'endpoint';
  if(unit.includes('site')) return 'site';
  if(unit.includes('TB')) return 'storage';
  if(unit.includes('mobile')) return 'mobile';
  if(unit.includes('company')) return 'company';
  return 'none';
}

// === Lane totals & sticky ===
function laneTotals(section){
  const arr = cards.filter(c=>c.section===section);
  let cost=0, rev=0, count=0; arr.forEach(c=>{ const qty=c.qty||0; if(qty&&c.cost!=null) cost+=qty*c.cost; if(qty&&c.price!=null) rev+=qty*c.price; if(qty) count++; });
  const profit=rev-cost; const margin=rev>0?((rev-cost)/rev*100):null; return {cost, rev, profit, margin, count};
}

function recalc(){
  let grandCost=0, grandRev=0;
  const users = parseInt(document.getElementById('primaryUsers').value)||10;
  let packagePrice=null;
  if(selectedPackage){ const pkg=PRESET_PACKAGES.find(p=>p.id===selectedPackage); if(pkg){ const sites=parseInt(document.getElementById('primarySites').value)||1; packagePrice=0; if(pkg.pricePerUser) packagePrice+=users*pkg.pricePerUser; if(pkg.siteFee) packagePrice+=sites*pkg.siteFee; if(pkg.minMonthly) packagePrice=Math.max(packagePrice,pkg.minMonthly); }}
  const overridePrice = toNumber(document.getElementById('primaryPackagePrice').value); if(overridePrice!=null && overridePrice>0) packagePrice=overridePrice;

  lanes.forEach(id=>{
    const t = laneTotals(id); grandCost+=t.cost; grandRev+= (packagePrice!=null? (id==='quote'?packagePrice:0) : t.rev);
    const badge = document.getElementById(`badge-${id}`); if(badge) badge.textContent = cards.filter(c=>c.section===id).length;
    const totalsEl=document.getElementById(`totals-${id}`);
    if(totalsEl && t.count>0){ totalsEl.innerHTML = `
      <div class="lane-total-item"><span class="lane-total-label">Cost:</span><span class="lane-total-value">${fmtCurrency(t.cost)}</span></div>
      <div class="lane-total-item"><span class="lane-total-label">Revenue:</span><span class="lane-total-value">${fmtCurrency(packagePrice!=null && id==='quote'?packagePrice:t.rev)}</span></div>
      <div class="lane-total-item"><span class="lane-total-label">Profit:</span><span class="lane-total-value">${fmtCurrency(packagePrice!=null && id==='quote'?packagePrice - t.cost : t.profit)}</span></div>
      <div class="lane-total-item"><span class="lane-total-label">Margin:</span><span class="lane-total-value">${(packagePrice!=null && id==='quote')? fmt((packagePrice - t.cost)/packagePrice*100)+"%" : (t.margin==null?"—":fmt(t.margin)+"%")}</span></div>`; totalsEl.style.display='grid';
    } else if(totalsEl){ totalsEl.style.display='none'; }
  });

  if(packagePrice!=null) grandRev=packagePrice;
  const clientCount = cards.filter(c=>c.section.startsWith('client')).length;
  const quoteCount = cards.filter(c=>c.section==='quote').length;
  document.getElementById('count-client').textContent = `${clientCount} item${clientCount!==1?'s':''}`;
  document.getElementById('count-quote').textContent = `${quoteCount} item${quoteCount!==1?'s':''}`;
  const gProfit = grandRev-grandCost; const gMargin = grandRev>0? (gProfit/grandRev*100):0; const perUser = (users>0? grandRev/users:0);
  document.getElementById('stickyMRR').textContent = fmtCurrency(grandRev);
  document.getElementById('stickyCost').textContent = fmtCurrency(grandCost);
  document.getElementById('stickyProfit').textContent = fmtCurrency(gProfit);
  document.getElementById('stickyProfit').className = `total-value ${gProfit>=0?'positive':'negative'}`;
  document.getElementById('stickyMargin').textContent = fmt(gMargin)+'%';
  document.getElementById('stickyPerUser').textContent = fmtCurrency(perUser);
}
