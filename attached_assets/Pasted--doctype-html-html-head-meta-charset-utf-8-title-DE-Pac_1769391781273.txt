function money(n){ return "$" + (Math.round(n*100)/100).toFixed(2); }

// qty calc per service
function qtyForService(s, users, sites, tbPerUser, c1Sites){
  if(s.id==="controlone_org" || s.id==="wazuh" || s.id==="hudu") return 1;
  if(s.id==="controlone_site") return c1Sites;
  if(s.id==="wasabi") return users * tbPerUser;
  return (s.qtyPerUser||0) * users;
}

// revenue per plan
function revenueForPlan(plan, users, sites){
  let rev = 0;
  if(plan.pricePerUser>0) rev += users*plan.pricePerUser;
  if(plan.siteFee>0)      rev += Math.max(sites,1)*plan.siteFee;
  if(plan.minMonthly>0)   rev  = Math.max(rev, plan.minMonthly);
  return rev;
}

function render(){
  const users   = parseInt(elUsers.value)||0;
  const sites   = parseInt(elSites.value)||0;
  const tbUser  = parseFloat(elTB.value)||0;
  const c1Sites = parseInt(elC1.value)||0;
  const alloc   = elAlloc.value; // "cost" or "equal"

  grid.innerHTML = "";

  plans.forEach(plan=>{
    // build rows for services in this plan
    const services = (bundle[plan.id]||[]).map(id=>svc.find(x=>x.id===id)).filter(Boolean);
    const rows = services.map(s=>{
      const qty   = qtyForService(s, users, sites, tbUser, c1Sites);
      const costU = s.cost;
      const lineC = qty * costU;
      return { sid:s.id, name:s.name, unit:s.unit, qty, costU, lineC };
    });

    const totalCost = rows.reduce((a,b)=>a+b.lineC, 0);
    const revenue   = revenueForPlan(plan, users, sites);

    // allocation
    let allocs = [];
    if(alloc==="equal"){
      const share = rows.length>0 ? 1/rows.length : 0;
      allocs = rows.map(() => revenue*share);
    }else{ // cost-weighted
      if(totalCost>0) allocs = rows.map(r => revenue * (r.lineC/totalCost));
      else            allocs = rows.map(()=>0);
    }

    const enriched = rows.map((r,i)=>{
      const rev = allocs[i]||0;
      const prof = rev - r.lineC;
      const margin = rev>0 ? (prof/rev*100) : 0;
      return {...r, rev, prof, margin};
    });

    const planProfit = revenue - totalCost;
    const planMargin = revenue>0 ? (planProfit/revenue*100) : 0;

    // card DOM
    const card = document.createElement('section');
    card.className = "card";

    const h = document.createElement('header');
    h.innerHTML = `
      <div class="title">${plan.name}</div>
      <div class="badges">
        ${plan.pricePerUser ? `<span class="badge">$${plan.pricePerUser}/user</span>` : `<span class="badge">Flat/Min</span>`}
        ${plan.siteFee ? `<span class="badge">Site $${plan.siteFee}</span>` : ``}
        ${plan.minMonthly ? `<span class="badge">Min $${plan.minMonthly}/mo</span>` : ``}
      </div>
    `;
    card.appendChild(h);

    // table
    const tbl = document.createElement('table');
    tbl.innerHTML = `
      <thead>
        <tr>
          <th style="min-width:220px">Service</th>
          <th>Unit</th>
          <th>Qty</th>
          <th>Cost/Unit</th>
          <th>Line Cost</th>
          <th>Allocated Revenue</th>
          <th>Profit</th>
          <th>Margin</th>
        </tr>
      </thead>
      <tbody></tbody>
    `;
    const tb = tbl.querySelector('tbody');

    enriched.forEach(row=>{
      const tr = document.createElement('tr');
      tr.innerHTML = `
        <td>
          <div style="display:flex;flex-direction:column;gap:6px">
            <div>${row.name}</div>
            <div class="bar" title="Cost vs Revenue bar">
              <div style="width:${row.rev>0 ? Math.min(100, (row.lineC/row.rev)*100).toFixed(1) : 0}%"></div>
            </div>
            <div class="mini">
              Cost: ${money(row.lineC)} · Rev: ${money(row.rev)} · Profit: ${money(row.prof)}
            </div>
          </div>
        </td>
        <td>${row.unit}</td>
        <td>${row.qty.toFixed(2)}</td>
        <td contenteditable="true" data-sid="${row.sid}" class="editable">${money(row.costU)}</td>
        <td>${money(row.lineC)}</td>
        <td>${money(row.rev)}</td>
        <td style="color:${row.prof>=0?'var(--good)':'var(--bad)'}">${money(row.prof)}</td>
        <td>${isFinite(row.margin)?row.margin.toFixed(2):'0.00'}%</td>
      `;
      tb.appendChild(tr);
    });

    card.appendChild(tbl);

    // footer KPIs
    const foot = document.createElement('div');
    foot.className = "foot";
    foot.innerHTML = `
      <div class="kpi"><span class="muted">Users:</span> <strong>${users}</strong></div>
      <div class="kpi"><span class="muted">Sites:</span> <strong>${sites}</strong></div>
      <div class="kpi"><span class="muted">Plan Revenue:</span> <strong>${money(revenue)}</strong></div>
      <div class="kpi"><span class="muted">Plan Cost:</span> <strong>${money(totalCost)}</strong></div>
      <div class="kpi"><span class="muted">Plan Profit:</span> <strong style="color:${planProfit>=0?'var(--good)':'var(--bad)'}">${money(planProfit)}</strong></div>
      <div class="kpi"><span class="muted">Plan Margin:</span> <strong>${isFinite(planMargin)?planMargin.toFixed(2):'0.00'}%</strong></div>
      <div class="kpi"><span class="muted">Allocation:</span> <strong>${alloc==="cost"?"Cost-weighted":"Equal per service"}</strong></div>
    `;
    card.appendChild(foot);

    grid.appendChild(card);
  });
}

document.getElementById('exportCSV').addEventListener('click', ()=>{
  const users   = parseInt(elUsers.value)||0;
  const sites   = parseInt(elSites.value)||0;
  const tbUser  = parseFloat(elTB.value)||0;
  const c1Sites = parseInt(elC1.value)||0;
  const alloc   = elAlloc.value;

  let rows = [];
  plans.forEach(plan=>{
    const services = (bundle[plan.id]||[]).map(id=>svc.find(x=>x.id===id)).filter(Boolean);
    const rws = services.map(s=>{
      const qty   = qtyForService(s, users, sites, tbUser, c1Sites);
      const costU = s.cost;
      const lineC = qty * costU;
      return { sid:s.id, name:s.name, unit:s.unit, qty, costU, lineC };
    });
    const totalCost = rws.reduce((a,b)=>a+b.lineC,0);
    const revenue   = revenueForPlan(plan, users, sites);
    let allocs = [];
    if(alloc==="equal"){
      const share = rws.length>0 ? 1/rws.length : 0;
      allocs = rws.map(()=> revenue*share);
    }else{
      allocs = totalCost>0 ? rws.map(r=>revenue*(r.lineC/totalCost)) : rws.map(()=>0);
    }
    rows.push(["Plan", plan.name, "", "", "", "", "", ""]);
    rows.push(["Service","Unit","Qty","Cost/Unit","Line Cost","Allocated Revenue","Profit","Margin %"]);
    rws.forEach((r,i)=>{
      const rev = allocs[i]||0;
      const prof = rev - r.lineC;
      const margin = rev>0 ? (prof/rev*100) : 0;
      rows.push([r.name, r.unit, r.qty.toFixed(2), r.costU, r.lineC, rev, prof, margin.toFixed(2)]);
    });
    const planProfit = revenue - totalCost;
    const planMargin = revenue>0 ? (planProfit/revenue*100) : 0;
    rows.push(["Totals","","","", totalCost, revenue, planProfit, planMargin.toFixed(2)]);
    rows.push(["","","","","","","",""]);
  });

  const csv = rows.map(r=>r.map(x=>`"${String(x).replace(/"/g,'""')}"`).join(",")).join("\n");
  const blob = new Blob([csv], {type:"text/csv;charset=utf-8;"});
  const url  = URL.createObjectURL(blob);
  const a    = document.createElement("a");
  a.href = url; a.download = "DE_Packages_Itemization.csv"; a.click();
  URL.revokeObjectURL(url);
});
