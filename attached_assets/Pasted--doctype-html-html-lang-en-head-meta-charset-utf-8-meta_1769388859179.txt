  // ===== Read UI =====
  function read(){
    // clamp & validate all number inputs (adds is-error class if something was corrected)
    document.querySelectorAll('#de-netcalc-wu-match input[type=number]').forEach(el=>{
      const min = isFinite(+el.min) ? +el.min : -Infinity;
      const max = isFinite(+el.max) ? +el.max :  Infinity;
      let v = parseInt(el.value,10); if(!isFinite(v)) v = min;
      v = Math.max(min, Math.min(max, v));
      const corrected = String(v) !== el.value.trim();
      if (corrected) el.value = v;
      el.classList.toggle('is-error', corrected);
    });

    const v={
      cs: $('optCloud').checked, uvl: $('optCore').checked,
      sites: clampInt($('sites').value,1),
      users: clampInt($('users').value,0),

      csU: clampInt($('csUplinksPerSite').value,0),
      csW: clampInt($('csWarmSites').value,0),
      csIke: clampInt($('csIpsec').value,0),
      csIPs: clampInt($('csIPs').value,0),

      gwId: $('uvlGateway').value,
      term: $('uvlTerm').value,
      u10g: $('uvlTenG').checked,
      sw8: clampInt($('sw8').value,0), sw24: clampInt($('sw24').value,0),
      sw24s: clampInt($('sw24sfp').value,0), sw48: clampInt($('sw48').value,0),
      ap5: clampInt($('ap5').value,0), ap6: clampInt($('ap6').value,0),

      optRet: $('optRetention').checked,
      retAppl: clampInt($('retAppliances').value,0),
      retSeats: clampInt($('retSeats').value,0),

      siteMin: Number($('siteMin')?.value || 0),
      gmCore: Number($('gmCore')?.value || 75),
      gmAdd:  Number($('gmAdd')?.value || 70),
      gmRet:  Number($('gmRetention')?.value || 0),
    };

    if(v.retAppl===0) v.retAppl = v.sites;
    if(v.retSeats===0) v.retSeats = v.users;

    $('csCard').style.display = v.cs ? 'block' : 'none';
    $('uvlCard').style.display = v.uvl ? 'block' : 'none';
    if(!v.cs && !v.uvl){ $('optCloud').checked=true; v.cs=true; }

    return v;
  }

  // ===== Calculators =====
  function csSolo(v){
    const rows=[]; let cost=0, sell=0;

    const perUserSell = sellFromCost(C1.user, v.gmCore);
    const siteSell = v.siteMin>0 ? v.siteMin*v.sites : 0;
    if(siteSell>0){ sell += siteSell; rows.push(['Site foundation (sell)', `${v.sites} site(s)`, 0, siteSell]); }
    sell += perUserSell*v.users; rows.push(['Per-user (sell)', `${v.users} user(s) @ ${fmt(perUserSell)}`, 0, perUserSell*v.users]);

    cost += C1.cloud; rows.push(['Cloud instance (cost)','1×',C1.cloud,'—']);
    cost += C1.site*v.sites; rows.push(['Site licenses (cost)',`${v.sites} site(s)`,C1.site*v.sites,'—']);
    cost += C1.user*v.users; rows.push(['Endpoint users (cost)',`${v.users} user(s)`,C1.user*v.users,'—']);

    const add=(lbl,qty,unit,qtyNum)=>{ if(qtyNum<=0) return; const c=unit*qtyNum, s=sellFromCost(c,v.gmAdd); cost+=c; sell+=s; rows.push([lbl,qty,c,s]); };
    add('Extra SD-WAN uplinks',`${v.csU}/site × ${v.sites} site(s)`,C1.uplink,v.csU*v.sites);
    add('Warm spare (licenses)',`${v.csW} site(s)`,C1.warm,v.csW);
    add('External IPSec connectors',`${v.csIke}`,C1.ipsec,v.csIke);
    add('Additional cloud IPs',`${v.csIPs}`,C1.ip,v.csIPs);

    if(v.optRet){
      const cA=C1.retAppliance*v.retAppl, cS=C1.retSeat*v.retSeats;
      const sA=sellFromCost(cA,v.gmRet), sS=sellFromCost(cS,v.gmRet);
      cost+=cA+cS; sell+=sA+sS;
      rows.push(['Compliance logging (appliances)',`${v.retAppl}`,cA,sA]);
      rows.push(['Compliance logging (seats)',`${v.retSeats}`,cS,sS]);
    }
    return {rows,cost,sell};
  }

  function uvlSolo(v){
    const rows=[]; let cost=0, sell=0;
    const siteSell = v.siteMin>0 ? v.siteMin*v.sites : 0;
    if(siteSell>0){ sell += siteSell; rows.push(['Site foundation (sell)', `${v.sites} site(s)`, 0, siteSell]); }

    // CoreStack cost side
    const g = getGatewayById(v.gwId);
    const gwMonthly = (UVL_BUCKET[g.tier] ?? 0) * (TERM_FACTOR[v.term] ?? 1);
    if (gwMonthly>0){
      const c = gwMonthly * v.sites;
      cost += c; rows.push(['Gateway subs (cost)', `${v.sites} × ${g.label}`, c, '—']);
    }
    if(v.u10g){ const bump=UVL_10G*v.sites; cost+=bump; rows.push(['10G performance uplift (cost)',`${v.sites} site(s)`,bump,'—']); }
    const swPerSite=UVL_SW.sw8*v.sw8 + UVL_SW.sw24*v.sw24 + UVL_SW.sw24s*v.sw24s + UVL_SW.sw48*v.sw48;
    if(swPerSite>0){ const c=swPerSite*v.sites; cost+=c; rows.push(['Switch subs (cost)',`${v.sites} site(s)`,c,'—']); }
    const apPerSite=UVL_AP.ap5*v.ap5 + UVL_AP.ap6*v.ap6;
    if(apPerSite>0){ const c=apPerSite*v.sites; cost+=c; rows.push(['Wi-Fi AP subs (cost)',`${v.sites} site(s)`,c,'—']); }

    if(v.optRet){
      const cA=C1.retAppliance*v.retAppl, cS=C1.retSeat*v.retSeats;
      const sA=sellFromCost(cA,v.gmRet), sS=sellFromCost(cS,v.gmRet);
      cost+=cA+cS; sell+=sA+sS;
      rows.push(['Compliance logging (appliances)',`${v.retAppl}`,cA,sA]);
      rows.push(['Compliance logging (seats)',`${v.retSeats}`,cS,sS]);
    }
    return {rows,cost,sell};
  }

  function merged(v){

    if (v.cs){

      sell += perUserSell*v.users; rows.push(['Per-user (sell)', `${v.users} user(s) @ ${fmt(perUserSell)}`, 0, perUserSell*v.users]);

      cost += C1.site*v.sites; rows.push(['Site licenses (cost)',`${v.sites} site(s)`,C1.site*v.sites,'—']);
      cost += C1.user*v.users; rows.push(['Endpoint users (cost)',`${v.users} user(s)`,C1.user*v.users,'—']);
      const add=(lbl,qty,unit,qtyNum)=>{ if(qtyNum<=0) return; const c=unit*qtyNum, s=sellFromCost(c,v.gmAdd); cost+=c; sell+=s; rows.push([lbl,qty,c,s]); };
      add('Extra SD-WAN uplinks',`${v.csU}/site × ${v.sites} site(s)`,C1.uplink,v.csU*v.sites);
      add('Warm spare (licenses)',`${v.csW} site(s)`,C1.warm,v.csW);
      add('External IPSec connectors',`${v.csIke}`,C1.ipsec,v.csIke);
      add('Additional cloud IPs',`${v.csIPs}`,C1.ip,v.csIPs);

    if (v.uvl){
      const g = getGatewayById(v.gwId);

      if (gwMonthly>0){
        const c = gwMonthly * v.sites;
        cost += c; rows.push(['Gateway subs (cost)', `${v.sites} × ${g.label}`, c, '—']);
      }
      if(v.u10g){ const bump=UVL_10G*v.sites; cost+=bump; rows.push(['10G performance uplift (cost)',`${v.sites} site(s)`,bump,'—']); }
      const swPerSite=UVL_SW.sw8*v.sw8 + UVL_SW.sw24*v.sw24 + UVL_SW.sw24s*v.sw24s + UVL_SW.sw48*v.sw48;
      if(swPerSite>0){ const c=swPerSite*v.sites; cost+=c; rows.push(['Switch subs (cost)',`${v.sites} site(s)`,c,'—']); }
      const apPerSite=UVL_AP.ap5*v.ap5 + UVL_AP.ap6*v.ap6;
      if(apPerSite>0){ const c=apPerSite*v.sites; cost+=c; rows.push(['Wi-Fi AP subs (cost)',`${v.sites} site(s)`,c,'—']); }

    if(v.optRet){
      const cA=C1.retAppliance*v.retAppl, cS=C1.retSeat*v.retSeats;

    }

  }

  // ===== Render helpers =====
  function renderTable(tbodyId, rows){
    const tb = $(tbodyId);
    tb.innerHTML="";
    rows.forEach(r=>{
      const tr=document.createElement('tr');
      const c = s => isNaN(s) ? '—' : fmt(s);
      tr.innerHTML = `<td>${r[0]}</td><td>${r[1]}</td><td>${c(r[2])}</td><td>${c(r[3])}</td>`;
      tb.appendChild(tr);
    });
  }

  // Live region for accessibility
  const live = document.createElement('div'); live.className='sr-only'; live.setAttribute('aria-live','polite'); live.style.position='absolute'; live.style.left='-9999px'; document.body.appendChild(live);
  function announceKPIs(){ live.textContent = `CloudShield ${$('kpiCs').textContent}, CoreStack ${$('kpiUvl').textContent}, Merged ${$('kpiMerged').textContent}, margin ${$('kpiGM').textContent}.`; }

  // ===== Compute with tiny debounce =====
  let _needs=false,_last=0;
  function scheduleCompute(){
    _last=Date.now();
    if(_needs) return;
    _needs=true;
    requestAnimationFrame(()=>{
      if(Date.now()-_last<24){ _needs=false; return scheduleCompute(); }
      _needs=false;
      updateGwInfo();
      const v = read();
      const cs = v.cs ? csSolo(v) : {rows:[],cost:0,sell:0};
      const uv = v.uvl ? uvlSolo(v) : {rows:[],cost:0,sell:0};
      const mg = (v.cs || v.uvl) ? merged(v) : {rows:[],cost:0,sell:0};

      $('kpiCs').textContent     = fmt(cs.sell);
      $('kpiUvl').textContent    = fmt(uv.sell);
      $('kpiMerged').textContent = fmt(mg.sell);
      $('kpiGM').textContent     = mg.sell>0 ? pc((mg.sell-mg.cost)/mg.sell) : '0%';

      renderTable('bdCs', cs.rows);     $('bdCsSum').textContent = fmt(cs.sell);
      renderTable('bdUvl', uv.rows);    $('bdUvlSum').textContent = fmt(uv.sell);
      renderTable('bdMerged', mg.rows); $('bdCost').textContent   = fmt(mg.cost); $('bdSell').textContent = fmt(mg.sell);
      announceKPIs();
    });
  }

  // ===== Admin lock =====
  function lockAdmin(){
    $('adminContent').hidden = true;
    $('adminLocked').hidden  = false;
    $('lockBadge').textContent = 'Locked';
  }
  function unlockAdmin(){
    $('adminContent').hidden = false;
    $('adminLocked').hidden  = true;
    $('lockBadge').textContent = 'Unlocked';
  }
  function askUnlock(){
    const dlg = $('pwDialog'); const input = $('adminPw');
    input.value = '';
    dlg.showModal();
    setTimeout(()=>input.focus(), 50);
  }
  function tryUnlock(){
    const ok = $('adminPw').value === ADMIN_PASSWORD;
    if(!ok){
      $('adminPw').classList.add('is-error');
      setTimeout(()=>$('adminPw').classList.remove('is-error'), 500);
      return;
    }
    $('pwDialog').close();
    unlockAdmin();
  }

  // ===== Wiring =====
  function wire(){
    populateGateways();

    document.querySelectorAll('#de-netcalc-wu-match input, #de-netcalc-wu-match select')
      .forEach(el=>{ el.addEventListener('input', scheduleCompute); el.addEventListener('change', scheduleCompute); });

    $('btnReset').addEventListener('click', ()=>{
      $('optCloud').checked = true; $('optCore').checked = true;
      $('sites').value = 1; $('users').value = 20;
      $('csUplinksPerSite').value = 0; $('csWarmSites').value = 0; $('csIpsec').value = 0; $('csIPs').value = 0;

      $('uvlGateway').value = GATEWAYS[0].id; $('uvlTerm').value = 'mo'; $('uvlTenG').checked = false;
      $('sw8').value=0; $('sw24').value=0; $('sw24sfp').value=0; $('sw48').value=0; $('ap5').value=0; $('ap6').value=0;

      $('optRetention').checked=false; $('retAppliances').value=0; $('retSeats').value=0;

      $('siteMin').value=0; $('gmCore').value=75; $('gmAdd').value=70; $('gmRetention').value=0;

      lockAdmin();
      scheduleCompute();
    });

    $('btnPrint').addEventListener('click', ()=>window.print());

    // Admin lock handlers
    $('btnUnlock').addEventListener('click', askUnlock);
    $('pwOk').addEventListener('click', tryUnlock);
    $('pwCancel').addEventListener('click', ()=>$('pwDialog').close());
    $('adminPw').addEventListener('keydown', e=>{ if(e.key==='Enter') tryUnlock(); });

    // First render
    lockAdmin();
    scheduleCompute();
  }

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', wire);
  else wire();
})();
