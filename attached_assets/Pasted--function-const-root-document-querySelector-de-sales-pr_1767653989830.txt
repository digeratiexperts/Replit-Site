(function(){
  const root = document.querySelector('#de-sales-process.de-sp4');
  if(!root) return;

  const tabs = Array.from(root.querySelectorAll('.de-sp4-tab'));
  const panels = Array.from(root.querySelectorAll('.de-sp4-panel'));
  const searchInput = root.querySelector('#deSp4Search');
  const progressBar = root.querySelector('#deSp4Progress');

  function getActivePanel(){
    return root.querySelector('.de-sp4-panel.is-active') || root.querySelector('.de-sp4-panel[data-panel="ecosystem"]');
  }

  function updateProgress(){
    const panel = getActivePanel();
    if(!panel || !progressBar) return;
    const cards = panel.querySelectorAll('[data-card]');
    const active = panel.querySelectorAll('[data-card].is-active');
    const pct = cards.length ? (active.length / cards.length) * 100 : 0;
    progressBar.style.width = `${pct}%`;
  }

  function activateTab(key){
    tabs.forEach(t=>{
      const on = t.dataset.tab === key;
      t.classList.toggle('is-active', on);
      t.setAttribute('aria-selected', on ? 'true' : 'false');
    });
    panels.forEach(p=>{
      const on = p.dataset.panel === key;
      p.hidden = !on;
      p.classList.toggle('is-active', on);
    });

    // Clear search filter on tab switch (more predictable)
    if(searchInput) searchInput.value = '';
    root.querySelectorAll('[data-card]').forEach(card=>{
      card.style.opacity = '1';
      card.style.transform = '';
    });

    updateProgress();
  }

  tabs.forEach(t => t.addEventListener('click', ()=>activateTab(t.dataset.tab)));

  function openDrawer(panel, card){
    const drawer = panel.querySelector('.de-sp4-drawer');
    if(!drawer) return;

    panel.querySelectorAll('[data-card]').forEach(c=>c.classList.remove('is-active'));
    card.classList.add('is-active');

    const title = card.querySelector('.de-sp4-cardtitle')?.textContent?.trim() || 'Details';
    const phase = card.getAttribute('data-phase') || 'Details';
    const meetingType = card.getAttribute('data-meeting-type') || '—';

    const details = card.querySelector('.de-sp4-details');
    drawer.querySelector('.de-sp4-drawer-title').textContent = title;
    drawer.querySelector('.de-sp4-drawer-tag').textContent = phase;

    // Inject meeting type at top of drawer body (always)
    const meetingTypeHtml = `<p><b>Meeting Type:</b> ${meetingType}</p>`;
    drawer.querySelector('.de-sp4-drawer-body').innerHTML =
      meetingTypeHtml + (details ? details.innerHTML : '<p>Add a hidden .de-sp4-details block in this card.</p>');

    drawer.hidden = false;
    drawer.scrollIntoView({behavior:'smooth', block:'nearest'});
    updateProgress();
  }

  root.addEventListener('click', (e)=>{
    const btn = e.target.closest('[data-toggle]');
    if(btn){
      const card = btn.closest('[data-card]');
      const panel = btn.closest('.de-sp4-panel');
      // Lead-gen row isn't a panel; allow it too by opening nearest panel drawer? For simplicity, do nothing.
      if(card && panel){
        e.preventDefault();
        openDrawer(panel, card);
      }
      return;
    }

    const x = e.target.closest('.de-sp4-drawer-x');
    if(x){
      const panel = x.closest('.de-sp4-panel');
      const drawer = panel?.querySelector('.de-sp4-drawer');
      if(drawer) drawer.hidden = true;
    }
  });

  // Search (current active panel only)
  if(searchInput){
    searchInput.addEventListener('input', (e)=>{
      const query = (e.target.value || '').trim().toLowerCase();
      const panel = getActivePanel();
      if(!panel) return;

      const cards = panel.querySelectorAll('[data-card]');
      cards.forEach(card=>{
        const keywords = (card.getAttribute('data-keywords') || '').toLowerCase();
        const meetingType = (card.getAttribute('data-meeting-type') || '').toLowerCase();
        const text = (card.textContent || '').toLowerCase();

        const matches =
          !query ||
          text.includes(query) ||
          keywords.includes(query) ||
          meetingType.includes(query);

        card.style.opacity = matches ? '1' : '0.25';
        card.style.transform = matches ? '' : 'scale(0.98)';
      });
    });
  }

  // Esc closes drawer
  document.addEventListener('keydown', (e)=>{
    if(e.key !== 'Escape') return;
    const panel = getActivePanel();
    const drawer = panel?.querySelector('.de-sp4-drawer');
    if(drawer && !drawer.hidden) drawer.hidden = true;
  });

  // init
  activateTab('ecosystem');
  updateProgress();
})();
