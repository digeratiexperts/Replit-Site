<script>
(function() {
  'use strict';
  
  if (window.itCalcInitialized) return;
  window.itCalcInitialized = true;

  const widget = document.getElementById('itcalcWidget');
  if (!widget) return;

  // Tab Switching
  const tabs = widget.querySelectorAll('.calc-tab');
  const contents = widget.querySelectorAll('.calc-content');
  
  tabs.forEach(tab => {
    tab.addEventListener('click', function() {
      const targetTab = this.dataset.tab;
      tabs.forEach(t => t.classList.remove('active'));
      contents.forEach(c => c.classList.remove('active'));
      this.classList.add('active');
      widget.querySelector('#' + targetTab).classList.add('active');
    });
  });

  // DOWNTIME CALCULATOR
  const dtAdvToggle = widget.querySelector('#dtAdvToggle');
  const dtAdvPanel = widget.querySelector('#dtAdvPanel');
  const dtCalcBtn = widget.querySelector('#dtCalcBtn');
  const dtResults = widget.querySelector('#dtResults');

  dtAdvToggle.addEventListener('click', function() {
    dtAdvPanel.classList.toggle('hidden');
    this.classList.toggle('open');
  });

  dtCalcBtn.addEventListener('click', function() {
    const industry = parseFloat(widget.querySelector('#dtIndustry').value);
    const employees = parseInt(widget.querySelector('#dtEmployees').value || 0);
    const wage = parseFloat(widget.querySelector('#dtWage').value || 0);
    const hours = parseFloat(widget.querySelector('#dtHours').value || 0);
    const incidents = parseInt(widget.querySelector('#dtIncidents').value || 1);
    const rto = parseFloat(widget.querySelector('#dtRto').value || 0);
    const rpo = parseFloat(widget.querySelector('#dtRpo').value || 0);
    const revenue = parseFloat(widget.querySelector('#dtRevenue').value || 0);
    const revenueImpact = parseFloat(widget.querySelector('#dtRevenueImpact').value || 0) / 100;

    const laborCost = employees * wage * hours;
    const multipliedCost = laborCost * industry;
    const revenueLoss = revenue > 0 ? revenue * hours * revenueImpact : 0;
    const perIncident = multipliedCost + revenueLoss;
    const annual = perIncident * incidents;
    const dataRisk = employees * wage * rpo * industry;

    widget.querySelector('#dtPerIncident').textContent = '$' + perIncident.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});
    widget.querySelector('#dtAnnual').textContent = '$' + annual.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});
    widget.querySelector('#dtDataRisk').textContent = '$' + dataRisk.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});

    const badge = widget.querySelector('#dtBadge');
    if (hours <= rto) {
      badge.className = 'calc-badge success';
      badge.textContent = '✓ Meeting RTO Target';
    } else if (hours <= rto * 1.5) {
      badge.className = 'calc-badge warn';
      badge.textContent = '⚠ Approaching RTO Limit';
    } else {
      badge.className = 'calc-badge danger';
      badge.textContent = '✕ Exceeding RTO Target';
    }

    dtResults.classList.remove('hidden');
    dtResults.classList.add('show');
    dtResults.scrollIntoView({behavior: 'smooth', block: 'nearest'});
  });

  // SERVICE COST CALCULATOR
  const svcAdvToggle = widget.querySelector('#svcAdvToggle');
  const svcAdvPanel = widget.querySelector('#svcAdvPanel');
  const svcCalcBtn = widget.querySelector('#svcCalcBtn');
  const svcResults = widget.querySelector('#svcResults');

  svcAdvToggle.addEventListener('click', function() {
    svcAdvPanel.classList.toggle('hidden');
    this.classList.toggle('open');
  });

  svcCalcBtn.addEventListener('click', function() {
    const employees = parseInt(widget.querySelector('#svcEmployees').value || 0);
    const packageType = widget.querySelector('#svcPackage').value;
    const mobileDevices = parseInt(widget.querySelector('#svcMobileDevices').value || 0);
    const mdmPackage = widget.querySelector('#mdmPackage').value;
    const quarterlyDisc = widget.querySelector('#quarterlyDiscount').checked;
    const sites = parseInt(widget.querySelector('#svcSites').value || 1);

    let perUserCost = 180;
    if (packageType === 'workplace') perUserCost = 250;
    if (packageType === 'business') perUserCost = 350;

    let totalCost = 0;
    if (employees <= 0) {
      totalCost = 0;
    } else if (employees <= 4) {
      totalCost = 700;
    } else {
      const extraEndpoints = Math.max(0, employees - 5);
      totalCost = 1200 + (extraEndpoints * perUserCost);
    }

    let mdmCost = 0;
    if (mdmPackage === 'basic-mdm') mdmCost = mobileDevices * 12.50;
    else if (mdmPackage === 'standard-mdm') mdmCost = mobileDevices * 22.50;
    else if (mdmPackage === 'compliance-mdm') mdmCost = mobileDevices * 35;

    totalCost += mdmCost;

    if (sites > 1) {
      totalCost *= (1 + (sites - 1) * 0.15);
    }

    const monthlyCost = totalCost;
    let quarterlyCost = monthlyCost * 3;
    if (quarterlyDisc) quarterlyCost *= 0.9;

    widget.querySelector('#svcMonthly').textContent = '$' + monthlyCost.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});
    widget.querySelector('#svcPerUser').textContent = '$' + (monthlyCost / Math.max(employees, 1)).toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});
    widget.querySelector('#svcQuarterly').textContent = '$' + quarterlyCost.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});

    svcResults.classList.remove('hidden');
    svcResults.classList.add('show');
    svcResults.scrollIntoView({behavior: 'smooth', block: 'nearest'});
  });
})();
</script>
